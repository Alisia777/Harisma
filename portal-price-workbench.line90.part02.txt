(value === null || value === undefined || value === "" || Number.isNaN(Number(value))) return "—";
    return `${(Number(value) * 100).toFixed(1)}%`;
  }

  function isoDate(value) {
    if (!value) return "";
    try {
      return String(value).slice(0, 10);
    } catch {
      return "";
    }
  }

  function marketLabel(key) {
    return ({ wb: "WB", ozon: "OZ", ym: "ЯМ" })[key] || key.toUpperCase();
  }

  function normalizeWorkbenchMarket(marketplace) {
    const normalized = String(marketplace || "").trim().toLowerCase();
    if (normalized === "ya") return "ym";
    return normalized || "wb";
  }

  function plusDaysLocal(days) {
    const base = new Date();
    base.setHours(12, 0, 0, 0);
    base.setDate(base.getDate() + days);
    return base.toISOString().slice(0, 10);
  }

  function meanOrNull(values) {
    const cleaned = values.filter((value) => value != null && !Number.isNaN(Number(value))).map(Number);
    if (!cleaned.length) return null;
    return cleaned.reduce((sum, value) => sum + value, 0) / cleaned.length;
  }

  function averageRecentMargin(items, limit = 3) {
    const values = (Array.isArray(items) ? items : [])
      .map((item) => parseNum(item?.marginPct))
      .filter((value) => value != null);
    if (!values.length) return null;
    const slice = values.slice(-limit);
    return slice.reduce((sum, value) => sum + value, 0) / slice.length;
  }

  function resolveAllowedMargin(row) {
    const support = row.supportRow || null;
    const avg3mActual = averageRecentMargin(support?.actualMonths, 3);
    const supportAllowed = parseNum(support?.allowedMarginPct);
    const seedAllowed = parseNum(row.allowedMarginPct);
    if (avg3mActual != null) {
      return {
        value: avg3mActual,
        source: "avg_3m_actual",
        note: "Средняя маржа последних 3 месяцев."
      };
    }
    if (supportAllowed != null) {
      return {
        value: supportAllowed,
        source: "support_workbook",
        note: "Допустимая маржа из рабочего файла."
      };
    }
    if (seedAllowed != null) {
      return {
        value: seedAllowed,
        source: "smart_seed",
        note: "Допустимая маржа из smart-пакета."
      };
    }
    return {
      value: 0.15,
      source: "default",
      note: "Резервное значение, потому что источник не заполнен."
    };
  }

  function defaultAllowedMargin(row) {
    return resolveAllowedMargin(row).value;
  }

  function parseStamp(value) {
    if (!value) return 0;
    const direct = Date.parse(value);
    if (Number.isFinite(direct)) return direct;
    const normalized = isoDate(value);
    if (!normalized) return 0;
    const parsed = Date.parse(`${normalized}T23:59:59Z`);
    return Number.isFinite(parsed) ? parsed : 0;
  }

  function seedFreshnessScore(payload) {
    if (!payload?.platforms) return 0;
    const scores = [parseStamp(payload.generatedAt)];
    Object.values(payload.platforms || {}).forEach((platform) => {
      ((platform || {}).rows || []).forEach((row) => {
        const score = parseStamp(row?.historyFreshnessDate);
        if (score) scores.push(score);
      });
    });
    return Math.max(...scores, 0);
  }

  function mergeSupportPayload(basePayload, overlayPayload) {
    const baseReady = basePayload?.platforms ? basePayload : null;
    const overlayReady = overlayPayload?.platforms ? overlayPayload : null;
    if (!baseReady && !overlayReady) return null;
    const merged = {
      ...(baseReady || {}),
      ...(overlayReady || {}),
      generatedAt: overlayReady?.generatedAt || baseReady?.generatedAt || "",
      platforms: {},
    };
    const platformKeys = new Set([
      ...Object.keys(baseReady?.platforms || {}),
      ...Object.keys(overlayReady?.platforms || {}),
    ]);
    platformKeys.forEach((platformKey) => {
      const basePlatform = (baseReady?.platforms || {})[platformKey] || {};
      const overlayPlatform = (overlayReady?.platforms || {})[platformKey] || {};
      const rows = { ...(basePlatform.rows || {}) };
      Object.entries(overlayPlatform.rows || {}).forEach(([articleKey, overlayRow]) => {
        rows[articleKey] = { ...(rows[articleKey] || {}), ...(overlayRow || {}) };
      });
      merged.platforms[platformKey] = {
        ...basePlatform,
        ...overlayPlatform,
        rows,
      };
    });
    return merged;
  }

  function chooseSeedPayload(snapshotPayload, localPayload) {
    const snapshotReady = snapshotPayload?.platforms ? snapshotPayload : null;
    const localReady = localPayload?.platforms ? localPayload : null;
    if (snapshotReady && localReady) {
      return seedFreshnessScore(localReady) >= seedFreshnessScore(snapshotReady)
        ? { payload: localReady, source: "local" }
        : { payload: snapshotReady, source: "snapshot" };
    }
    if (localReady) return { payload: localReady, source: "local" };
    if (snapshotReady) return { payload: snapshotReady, source: "snapshot" };
    return null;
  }

  function supportPlatformKey(marketplace) {
    return marketplace === "ya" ? "ym" : marketplace;
  }

  function latestSeriesDate(series, predicate) {
    const rows = Array.isArray(series) ? [...series] : [];
    for (let index = rows.length - 1; index >= 0; index -= 1) {
      if (predicate(rows[index])) return isoDate(rows[index]?.date);
    }
    return "";
  }

  function buildHistoryMeta(series, row) {
    const hasDailyFact = (Array.isArray(series) ? series : []).some((item) => {
      return parseNum(item?.ordersUnits) != null || parseNum(item?.deliveredUnits) != null || parseNum(item?.revenue) != null;
    });
    const latestPriceDate = latestSeriesDate(series, (item) => parseNum(item?.price) != null || parseNum(item?.clientPrice) != null);
    const latestDailyDate = latestSeriesDate(series, (item) => parseNum(item?.ordersUnits) != null || parseNum(item?.deliveredUnits) != null || parseNum(item?.revenue) != null);
    if (hasDailyFact) {
      return {
        mode: "daily",
        freshnessDate: latestDailyDate || isoDate(row?.historyFreshnessDate),
        note: String(row?.historyNote || "").trim()
      };
    }
    if (latestPriceDate || row?.historyFreshnessDate) {
      return {
        mode: "price_only",
        freshnessDate: isoDate(row?.historyFreshnessDate) || latestPriceDate,
        note: String(row?.historyNote || "").trim()
      };
    }
    return {
      mode: "empty",
      freshnessDate: "",
      note: String(row?.historyNote || "").trim()
    };
  }

  function supportRow(marketplace, articleKey) {
    if (!articleKey) return null;
    const platformKey = supportPlatformKey(String(marketplace || "").trim().toLowerCase());
    return state.priceWorkbench.support?.platforms?.[platformKey]?.rows?.[articleKey] || null;
  }

  function mergeSupportRow(row) {
    const support = supportRow(row.marketplace, row.articleKey);
    if (!support) return { ...row, supportRow: null };
    const merged = { ...row, supportRow: support };
    const workbookAllowed = parseNum(support.allowedMarginPct);
    const profitability = parseNum(support.requiredPriceForProfitability);
    const hardMin = parseNum(support.historicalMinProfitablePrice) ?? parseNum(support.hardMinPrice) ?? parseNum(support.summary?.hardMinPrice) ?? parseNum(support.sellerSummary?.hardMinPrice);
    const zoneFrom = parseNum(support.workingZoneFrom) ?? parseNum(support.summary?.workingZoneFrom) ?? parseNum(support.sellerSummary?.workingZoneFrom);
    const zoneTo = parseNum(support.workingZoneTo) ?? parseNum(support.summary?.workingZoneTo) ?? parseNum(support.sellerSummary?.workingZoneTo);
    if (workbookAllowed != null) merged.allowedMarginPct = workbookAllowed;
    if (profitability != null) merged.requiredPriceForProfitability = profitability;
    if (hardMin != null) merged.hardMinPrice = hardMin;
    if (parseNum(support.minPrice) != null) merged.minPrice = parseNum(support.minPrice);
    if (parseNum(support.maxPrice) != null) merged.maxPrice = parseNum(support.maxPrice);
    if (zoneFrom != null) merged.workingZoneFrom = zoneFrom;
    if (zoneTo != null) merged.workingZoneTo = zoneTo;
    if (!merged.name && support.name) merged.name = support.name;
    if (!merged.owner && support.owner) merged.owner = support.owner;
    if (!merged.segment && support.segment) merged.segment = support.segment;
    if (!merged.abc && support.abc) merged.abc = support.abc;
    return merged;
  }

  function dateKeyToUtc(value) {
    const normalized = isoDate(value);
    if (!normalized) return null;
    const [year, month, day] = normalized.split("-").map((part) => Number(part));
    if (!year || !month || !day) return null;
    return new Date(Date.UTC(year, month - 1, day));
  }

  function monthBounds(monthKey) {
    const match = String(monthKey || "").match(/^(\d{4})-(\d{2})$/);
    if (!match) return null;
    const year = Number(match[1]);
    const monthIndex = Number(match[2]) - 1;
    const start = new Date(Date.UTC(year, monthIndex, 1));
    const end = new Date(Date.UTC(year, monthIndex + 1, 0));
    return { start, end, days: end.getUTCDate() };
  }
