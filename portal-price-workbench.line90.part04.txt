  function allDatesForMarket(market) {
    const seen = new Set();
    const dates = [];
    baseRowsForMarket(market).forEach((row) => {
      (row.monthly || []).forEach((item) => {
        const value = isoDate(item.date);
        if (value && !seen.has(value)) {
          seen.add(value);
          dates.push(value);
        }
      });
    });
    dates.sort();
    return dates;
  }

  function presetRange(preset, market) {
    const dates = allDatesForMarket(market);
    if (!dates.length) return { dateFrom: "", dateTo: "" };
    const min = dates[0];
    const max = dates[dates.length - 1];
    if (preset === "all") return { dateFrom: min, dateTo: max };
    const maxDate = new Date(max);
    if (preset === "prev_month") {
      const start = new Date(Date.UTC(maxDate.getUTCFullYear(), maxDate.getUTCMonth() - 1, 1));
      const end = new Date(Date.UTC(maxDate.getUTCFullYear(), maxDate.getUTCMonth(), 0));
      return { dateFrom: start.toISOString().slice(0, 10), dateTo: end.toISOString().slice(0, 10) };
    }
    if (preset === "month") {
      const start = new Date(Date.UTC(maxDate.getUTCFullYear(), maxDate.getUTCMonth(), 1));
      return { dateFrom: start.toISOString().slice(0, 10), dateTo: max };
    }
    const days = PRESETS[preset]?.days || 30;
    const start = new Date(maxDate);
    start.setUTCDate(start.getUTCDate() - (days - 1));
    const startIso = start.toISOString().slice(0, 10);
    return { dateFrom: startIso < min ? min : startIso, dateTo: max };
  }

  function ensureDateRange() {
    if (state.priceWorkbench.filters.dateFrom && state.priceWorkbench.filters.dateTo) return;
    const range = presetRange(state.priceWorkbench.filters.preset || "month", state.priceWorkbench.filters.market || "wb");
    state.priceWorkbench.filters.dateFrom = range.dateFrom;
    state.priceWorkbench.filters.dateTo = range.dateTo;
  }

  function applyPreset(preset) {
    state.priceWorkbench.filters.preset = preset;
    const range = presetRange(preset, state.priceWorkbench.filters.market || "wb");
    state.priceWorkbench.filters.dateFrom = range.dateFrom;
    state.priceWorkbench.filters.dateTo = range.dateTo;
    renderPriceWorkbench();
  }

  function withinRange(value) {
    const dateValue = isoDate(value);
    if (!dateValue) return false;
    const from = state.priceWorkbench.filters.dateFrom || "";
    const to = state.priceWorkbench.filters.dateTo || "";
    if (from && dateValue < from) return false;
    if (to && dateValue > to) return false;
    return true;
  }

  function mergeEntryLayers() {
    const map = new Map();
    (state.priceWorkbench.remoteEntries || []).forEach((item) => map.set(workbenchKey(item.marketplace, item.articleKey), item));
    (state.priceWorkbench.localEntries || []).forEach((item) => map.set(workbenchKey(item.marketplace, item.articleKey), item));
    return map;
  }

  function computePlanMetrics(row, series) {
    const rangeFrom = state.priceWorkbench.filters.dateFrom || "";
    const rangeTo = state.priceWorkbench.filters.dateTo || "";
    if (!rangeFrom || !rangeTo || rangeFrom > rangeTo) {
      return {
        planUnitsSelected: null,
        actualUnitsSelected: null,
        completionPct: null,
        selectedDays: 0,
        actualRevenueSelected: null,
        actualSource: "",
        planSource: "",
      };
    }
    const rangeStart = dateKeyToUtc(rangeFrom);
    const rangeEnd = dateKeyToUtc(rangeTo);
    if (!rangeStart || !rangeEnd) {
      return {
        planUnitsSelected: null,
        actualUnitsSelected: null,
        completionPct: null,
        selectedDays: 0,
        actualRevenueSelected: null,
        actualSource: "",
        planSource: "",
      };
    }
    const selectedDays = Math.floor((rangeEnd - rangeStart) / 86400000) + 1;
    const support = row.supportRow || null;
    const planFromSupport = sumMonthlyBuckets(support?.planMonths, rangeStart, rangeEnd);
    const fallbackPlanUnits = parseNum(row.planMonthUnits);
    const fallbackPlanDays = parseNum(row.planMonthDays) || 30;
    const planUnitsSelected = planFromSupport?.units != null
      ? planFromSupport.units
      : (fallbackPlanUnits && fallbackPlanDays ? (fallbackPlanUnits / fallbackPlanDays) * selectedDays : null);
    const actualPool = series.filter((item) => {
      const itemDate = isoDate(item.date);
      return itemDate && itemDate >= rangeFrom && itemDate <= rangeTo;
    });
    const actualValues = actualPool
      .map((item) => parseNum(item.ordersUnits) ?? parseNum(item.deliveredUnits))
      .filter((value) => value != null);
    const actualRevenueValues = actualPool
      .map((item) => parseNum(item.revenue))
      .filter((value) => value != null);
    let actualUnitsSelected = actualValues.length ? actualValues.reduce((sum, value) => sum + value, 0) : null;
    let actualRevenueSelected = actualRevenueValues.length ? actualRevenueValues.reduce((sum, value) => sum + value, 0) : null;
    let actualSource = actualUnitsSelected != null ? "daily" : "";
    if (actualUnitsSelected == null) {
      const monthlyActual = sumMonthlyBuckets(support?.actualMonths, rangeStart, rangeEnd);
      if (monthlyActual?.units != null) {
        actualUnitsSelected = monthlyActual.units;
        actualRevenueSelected = monthlyActual.revenue;
        actualSource = "monthly";
      }
    }
    const completionPct = actualUnitsSelected != null && planUnitsSelected > 0 ? actualUnitsSelected / planUnitsSelected : null;
    return {
      planUnitsSelected,
      actualUnitsSelected,
      completionPct,
      selectedDays,
      actualRevenueSelected,
      actualSource,
      planSource: planFromSupport?.units != null ? "monthly_plan" : (planUnitsSelected != null ? "seed" : ""),
    };
  }

  function compactHistoryNote(row, limit = 140) {
    const note = String(row?.historyNoteDisplay || row?.historyNote || "").trim();
    if (!note) return "";
    if (note.length <= limit) return note;
    return `${note.slice(0, limit - 1).trimEnd()}…`;
  }

  function supportStatusKind(value) {
    const raw = String(value || "").trim().toLowerCase();
    if (!raw) return "info";
    if (/актуал|продается|доставлен|в наличии/.test(raw)) return "ok";
    if (/вывод|архив|стоп|неактуал|не продается|скрыт/.test(raw)) return "danger";
    if (/пауза|огранич|нет данных|конфликт/.test(raw)) return "warn";
    return "info";
  }

  function productStatusValue(row) {
    const support = row?.supportRow || null;
    if (!support) return "";
    const values = [];
    if (support.productStatus) values.push(String(support.productStatus).trim());
    if (support.repricerStatus) {
      const repricer = `Repricer: ${String(support.repricerStatus).trim()}`;
      if (!values.includes(repricer)) values.push(repricer);
    }
    return values.filter(Boolean).join(" · ");
  }

  function productStatusNote(row) {
    const support = row?.supportRow || null;
    if (!support) return "";
    const values = [];
    if (support.inCurrentExport === false) values.push("Позиция не отмечена в текущем export.");
    if (support.inCurrentExport === true) values.push("Позиция есть в текущем export.");
    if (support.historyQuality) values.push(`Качество истории: ${String(support.historyQuality).trim()}`);
    if (support.statusInterpretation) values.push(String(support.statusInterpretation).trim());
    else if (support.buyerInterpretation) values.push(String(support.buyerInterpretation).trim());
    if (support.statusComment) values.push(String(support.statusComment).trim());
    return values.filter(Boolean).join(" · ");
  }

  function productStatusBadges(row) {

    if (!support) return [];

    if (support.repricerStatus) values.push(badge(`Repricer: ${String(support.repricerStatus).trim()}`, supportStatusKind(support.repricerStatus)));
    return values.filter(Boolean);
  }
