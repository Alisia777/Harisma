function priceSignalMarkup(row) {
    const items = signalsForRow(row);
    if (!items.length) return badge("Стабильно", "ok");
    return items.map((item) => badge(item.label, item.kind)).join("");
  }

  function pricePriorityRows(rows, limit = 6) {
    return rows
      .slice()
      .sort((left, right) => {
        return pricePriorityScore(right) - pricePriorityScore(left)
          || riskScore(right) - riskScore(left)
          || (Number(right.turnoverDisplay) || 0) - (Number(left.turnoverDisplay) || 0)
          || (Number(left.completionPctSelected) || 0) - (Number(right.completionPctSelected) || 0);
      })
      .slice(0, limit);
  }

  function renderSummary(rows) {
    const totals = aggregateMetrics(rows);
    const belowMargin = rows.filter((row) => row.marginTotalPct != null && row.effectiveAllowedMarginPct != null && row.marginTotalPct < row.effectiveAllowedMarginPct).length;
    const underPlan = rows.filter((row) => row.completionPctSelected != null && row.completionPctSelected < 0.8).length;
    const tasks = rows.filter((row) => row.needsTask || relatedTasks(row).length).length;
    const dailyReady = rows.filter((row) => row.historyHasDailyFact).length;
    const priceOnly = rows.filter((row) => row.historyMode === "price_only").length;
    const priority = rows.filter((row) => pricePriorityScore(row) >= 18).length;
    const avgTurnover = meanOrNull(rows.map((row) => row.turnoverDisplay));
    return `
      <div class="price-summary-grid">
        <div class="price-summary-card"><span class="label">В контуре</span><strong>${fmt.int(rows.length)}</strong><small>Артикулы выбранной площадки и периода, по которым можно принимать решение.</small></div>
        <div class="price-summary-card"><span class="label">Приоритет сегодня</span><strong>${fmt.int(priority)}</strong><small>Позиции, где уже совпали несколько сигналов риска и их стоит разбирать первыми.</small></div>
        <div class="price-summary-card"><span class="label">Ниже допустимой маржи</span><strong>${fmt.int(belowMargin)}</strong><small>Сравнение идет с допустимой маржой, которую берем как среднюю за 3 месяца, если она есть в поддерживающих файлах.</small></div>
        <div class="price-summary-card"><span class="label">Под риском по плану</span><strong>${fmt.int(underPlan)}</strong><small>${totals.planUnits == null ? "План по SKU еще не загружен для выбранного диапазона." : `Суммарный план по видимым SKU ${fmt.int(totals.planUnits)} шт.`}</small></div>
        <div class="price-summary-card"><span class="label">Выручка / средний чек</span><strong>${moneyShort(totals.periodRevenue)}</strong><small>${totals.avgCheck == null ? "Средний чек появится там, где есть дневной факт." : `Средний чек ${moneyShort(totals.avgCheck)}.`}</small></div>
        <div class="price-summary-card"><span class="label">Средняя оборачиваемость</span><strong>${avgTurnover == null ? "—" : `${fmt.num(avgTurnover, 1)} дн.`}</strong><small>Показываем средний уровень по видимому списку, чтобы быстро понять скорость оборота площадки.</small></div>
        <div class="price-summary-card"><span class="label">История по дням</span><strong>${fmt.int(dailyReady)}</strong><small>${priceOnly ? `Еще у ${fmt.int(priceOnly)} SKU история только по цене и оборачиваемости.` : "По всем видимым позициям история доступна на уровне дня."}</small></div>
        <div class="price-summary-card"><span class="label">Задачи и решения</span><strong>${fmt.int(tasks)}</strong><small>Позиции, где уже есть задача, согласование или ручное решение по цене.</small></div>
      </div>
    `;
  }
