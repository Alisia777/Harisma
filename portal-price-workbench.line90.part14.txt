  function renderPriceWorkbench() {
    const root = document.getElementById("view-prices");
    if (!root) return;
    ensureStyles();
    ensurePriceManagementStyles();
    ensureBranding();
    if (state.priceWorkbench.loading && !state.priceWorkbench.seed) {
      root.innerHTML = `<div class="card"><div class="price-empty">Загружаю контур цен…</div></div>`;
      return;
    }
    if (state.priceWorkbench.issue && !state.priceWorkbench.seed) {
      root.innerHTML = `<div class="card"><div class="price-empty">Не удалось загрузить блок цен: ${escapeHtml(state.priceWorkbench.issue)}</div></div>`;
      return;
    }
    if (!state.priceWorkbench.seed) {
      root.innerHTML = `<div class="card"><div class="price-empty">Ценовой контур еще не загружен.</div></div>`;
      return;
    }
    ensureDateRange();
    const rows = filteredRows();
    selectedRow(rows);
    root.innerHTML = `
      <div class="price-workbench-shell">
        ${renderToolbar()}
        ${renderSummary(rows)}
        ${renderPriorityQueue(rows)}
        <div class="price-note">Площадка и календарь влияют на рабочий список, план-факт, историю по дням и карточку артикула. Daily-факт приходит через bridge из Google Sheets, который обновляется около 09:00 МСК. По Ozon дневной факт в текущем пакете до 16 апреля 2026, по WB после 31 марта 2026 часть SKU пока идет без daily-продаж, поэтому это отдельно подсвечено в таблице и карточке.</div>
        ${renderTable(rows)}
      </div>
    `;
    const bind = (selector, eventName, handler) => {
      const node = root.querySelector(selector);
      if (node) node.addEventListener(eventName, handler);
    };
    bind("#priceSearch", "input", (event) => { state.priceWorkbench.filters.search = event.target.value; renderPriceWorkbench(); });
    bind("#priceOwnerFilter", "change", (event) => { state.priceWorkbench.filters.owner = event.target.value; renderPriceWorkbench(); });
    bind("#priceStatusFilter", "change", (event) => { state.priceWorkbench.filters.status = event.target.value; renderPriceWorkbench(); });
    bind("#priceSortMode", "change", (event) => { state.priceWorkbench.filters.sort = event.target.value; renderPriceWorkbench(); });
    bind("#priceRiskOnly", "change", (event) => { state.priceWorkbench.filters.riskOnly = event.target.checked; renderPriceWorkbench(); });
    bind("#priceDateFrom", "change", (event) => { state.priceWorkbench.filters.dateFrom = event.target.value; state.priceWorkbench.filters.preset = "custom"; renderPriceWorkbench(); });
    bind("#priceDateTo", "change", (event) => { state.priceWorkbench.filters.dateTo = event.target.value; state.priceWorkbench.filters.preset = "custom"; renderPriceWorkbench(); });
    root.querySelectorAll("[data-price-preset]").forEach((button) => button.addEventListener("click", () => applyPreset(button.dataset.pricePreset)));
    root.querySelectorAll("[data-price-market]").forEach((button) => button.addEventListener("click", () => {
      state.priceWorkbench.filters.market = button.dataset.priceMarket;
      state.priceWorkbench.selectedKey = "";
      const range = presetRange(state.priceWorkbench.filters.preset || "month", state.priceWorkbench.filters.market);
      state.priceWorkbench.filters.dateFrom = range.dateFrom;
      state.priceWorkbench.filters.dateTo = range.dateTo;
      renderPriceWorkbench();
    }));
    root.querySelectorAll("[data-price-open]").forEach((rowNode) => rowNode.addEventListener("click", () => {
      state.priceWorkbench.selectedKey = rowNode.dataset.priceOpen;
      const row = getCurrentRowByKey(rowNode.dataset.priceOpen);
      renderPriceWorkbench();
      if (row) renderPriceModal(row);
    }));
  }

  async function openPriceWorkbenchArticle(options) {
    const articleKey = String(options?.articleKey || "").trim();
    if (!articleKey) return;
    const market = normalizeWorkbenchMarket(options?.marketplace || state.priceWorkbench.filters.market || "wb");
    await ensureSeedLoaded();
    state.priceWorkbench.filters.market = market;
    if (options?.dateFrom) state.priceWorkbench.filters.dateFrom = isoDate(options.dateFrom);
    if (options?.dateTo) state.priceWorkbench.filters.dateTo = isoDate(options.dateTo);
    if (options?.dateFrom || options?.dateTo) state.priceWorkbench.filters.preset = "custom";
    state.priceWorkbench.selectedKey = workbenchKey(market, articleKey);
    if (typeof setView === "function") setView("prices");
    else renderPriceWorkbench();
    const row = getCurrentRowByKey(state.priceWorkbench.selectedKey);
    if (row) renderPriceModal(row);
  }

  window.renderPriceWorkbench = renderPriceWorkbench;
  window.ensurePriceWorkbenchSeedLoaded = ensureSeedLoaded;
  window.openPriceWorkbenchArticle = openPriceWorkbenchArticle;

  const prevPullRemoteState = pullRemoteState;
  pullRemoteState = async function (rerender = true) {
    await prevPullRemoteState(false);
    state.priceWorkbench.remoteReady = false;
    state.priceWorkbench.remoteDisabled = false;
    await ensureRemoteLoaded();
    if (rerender) rerenderCurrentView();
  };

  const prevPushStateToRemote = pushStateToRemote;
  pushStateToRemote = async function () {
    await prevPushStateToRemote();
    if (state.team.client && !state.priceWorkbench.remoteDisabled && state.priceWorkbench.localEntries?.length) {
      try {
        await upsertRemoteEntries(state.priceWorkbench.localEntries);
        const unsynced = (state.priceWorkbench.localHistory || []).filter((item) => !item.synced);
        if (unsynced.length) {
          try {
            await insertRemoteHistory(unsynced);
            unsynced.forEach((item) => { item.synced = true; });
            saveLocalState();
          } catch (historyError) {
            console.error(historyError);
          }
        }
      } catch (error) {
        console.error(error);
      }
    }
    rerenderCurrentView();
  };

  const prevRerenderCurrentView = rerenderCurrentView;
  rerenderCurrentView = function () {
    prevRerenderCurrentView();
    ensureBranding();
    if ((state.activeView || "dashboard") === "prices") {
      renderPriceWorkbench();
      ensureSeedLoaded();
      ensureRemoteLoaded();
    }
  };

  loadLocalState();
  ensureStyles();
  ensureBranding();
})();
